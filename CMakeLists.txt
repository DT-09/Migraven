cmake_minimum_required(VERSION 3.16)
project(migraven VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(MIGRAVEN_BUILD_TESTS "Build Migraven tests" ON)
add_library(migraven_core STATIC src/csv.c src/jsonl.c src/compare.c src/util.c)
target_include_directories(migraven_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(migraven_core PRIVATE /W4)
else()
  target_compile_options(migraven_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_executable(migraven src/main.c)
target_link_libraries(migraven PRIVATE migraven_core)
if(MIGRAVEN_BUILD_TESTS)
  enable_testing()
  add_executable(migraven_tests tests/test_core.c)
  target_link_libraries(migraven_tests PRIVATE migraven_core)
  add_test(NAME migraven_core_tests COMMAND migraven_tests)
  find_package(Python3 3.10 COMPONENTS Interpreter REQUIRED)
  add_test(NAME migraven_cli_tests
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:migraven>)
  add_test(NAME migraven_csv_edge_tests
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_csv_edges.py $<TARGET_FILE:migraven>)
  add_test(NAME migraven_report_tests
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_report.py)
  add_test(NAME migraven_evidence_tests
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_evidence.py)
  add_test(NAME migraven_demo_tests
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_demo.py $<TARGET_FILE:migraven>)
  add_test(NAME migraven_package_tests
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_package.py $<TARGET_FILE:migraven>)
  add_test(NAME migraven_website_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_website.py)
endif()
install(TARGETS migraven migraven_core)
install(DIRECTORY include/ DESTINATION include)
